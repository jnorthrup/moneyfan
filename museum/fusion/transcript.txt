<script>
/******************************************************************************************
 * 1.  Market & Contract Specs
 ******************************************************************************************/
const INITIAL_PRICE = 50000;        // BTC-USD
const VOLATILITY    = 0.45;         // 45 % annual
const RISK_FREE     = 0.03;         // 3 % annual
const STRIKE        = INITIAL_PRICE;
const DAYS_TO_EXP   = 7;
const SECONDS_PER_DAY = 24*60*60;

/******************************************************************************************
 * 2.  Tiny Utils
 ******************************************************************************************/
const randn = () => { let u=0,v=0; while(u===0)u=Math.random(); while(v===0)v=Math.random(); return Math.sqrt(-2*Math.log(u))*Math.cos(2*Math.PI*v); };
const blackScholes = (S,K,T,sigma,r,call=true) => {
    const d1 = (Math.log(S/K) + (r + 0.5*sigma*sigma)*T) / (sigma*Math.sqrt(T));
    const d2 = d1 - sigma*Math.sqrt(T);
    const nd = x => 0.5*(1+erf(x/Math.sqrt(2)));
    if (call) return S*nd(d1) - K*Math.exp(-r*T)*nd(d2);
    else      return K*Math.exp(-r*T)*nd(-d2) - S*nd(-d1);
};
const delta = (S,K,T,sigma,r,call=true) => {
    const d1 = (Math.log(S/K) + (r + 0.5*sigma*sigma)*T) / (sigma*Math.sqrt(T));
    const nd = x => 0.5*(1+erf(x/Math.sqrt(2)));
    return call ? nd(d1) : nd(d1)-1;
};

/******************************************************************************************
 * 3.  Bot Base Class
 ******************************************************************************************/
class Bot {
    constructor(name, color, cash=1e6) {
        this.name = name;
        this.color = color;
        this.cash = cash;
        this.shares = 0;
        this.options = 0;          // net position (+=long,-=short)
        this.premium = 0;          // total premium paid/received
        this.log = [];
        this.arenaDiv = null;
    }
    logMsg(msg) { this.log.push(msg); if(this.log.length>12)this.log.shift(); this.render(); }
    render() {
        if (!this.arenaDiv) return;
        this.arenaDiv.innerHTML = `
<h3 style="color:${this.color}">${this.name}</h3>
<div class="log">Cash: ${this.cash.toFixed(0)} USD
Shares: ${this.shares.toFixed(2)}
Options: ${this.options.toFixed(2)}
PnL: ${(this.cash+this.shares*market.price+this.options*Math.max(market.price-STRIKE,0)-1e6).toFixed(0)}
${this.log.join('\n')}</div>`;
    }
    value() { return this.cash + this.shares*market.price + (this.options>0?this.options*Math.max(market.price-STRIKE,0):this.options*Math.min(market.price-STRIKE,0)); }
}

/******************************************************************************************
 * 4.  Concrete Strategies
 ******************************************************************************************/
class DeltaHedger extends Bot {
    think() {
        const T = Math.max(market.T, 1/secondsPerYear);
        const d = delta(market.price, STRIKE, T, VOLATILITY, RISK_FREE, true);
        const targetShares = -this.options * d;
        const deltaShares = targetShares - this.shares;
        if (Math.abs(deltaShares) < 0.01) return;
        if (deltaShares > 0) { // buy
            const qty = deltaShares;
            const cost = qty * market.price;
            if (cost <= this.cash) {
                this.cash -= cost; this.shares += qty;
                this.logMsg(`HEDGE BUY ${qty.toFixed(2)} @ ${market.price.toFixed(0)}`);
            }
        } else { // sell
            const qty = -deltaShares;
            this.cash += qty * market.price; this.shares -= qty;
            this.logMsg(`HEDGE SELL ${qty.toFixed(2)} @ ${market.price.toFixed(0)}`);
        }
    }
}
class VolArb extends Bot {
    constructor(name, color) { super(name, color); this.targetVol = VOLATILITY; }
    think() {
        const realized = market.volatility(); // 24h realized
        const T = Math.max(market.T, 1/secondsPerYear);
        const optPrice = blackScholes(market.price, STRIKE, T, this.targetVol, RISK_FREE, true);
        const fairPrice = blackScholes(market.price, STRIKE, T, realized, RISK_FREE, true);
        if (optPrice < fairPrice*0.97) { // buy cheap
            const qty = 10;
            if (this.cash > qty*optPrice) {
                this.options += qty; this.cash -= qty*optPrice; this.premium += qty*optPrice;
                this.logMsg(`VOL BUY 10 @ ${optPrice.toFixed(0)} (realVol ${(realized*100).toFixed(1)}%)`);
            }
        } else if (optPrice > fairPrice*1.03) { // sell rich
            const qty = 10;
            this.options -= qty; this.cash += qty*optPrice; this.premium -= qty*optPrice;
            this.logMsg(`VOL SELL 10 @ ${optPrice.toFixed(0)} (realVol ${(realized*100).toFixed(1)}%)`);
        }
    }
}
class Momentum extends Bot {
    think() {
        const ret = (market.price - market.history[Math.max(0,market.history.length-20)]) / market.history[Math.max(0,market.history.length-20)];
        if (ret > 0.02 && this.cash > 5000) {
            const qty = 1;
            const optPrice = blackScholes(market.price, STRIKE, Math.max(market.T,1/secondsPerYear), VOLATILITY, RISK_FREE, true);
            this.options += qty; this.cash -= optPrice;
            this.logMsg(`MOMO CALL ${qty} @ ${optPrice.toFixed(0)}`);
        }
        if (ret < -0.02 && this.options > 1) {
            const qty = 1;
            const optPrice = blackScholes(market.price, STRIKE, Math.max(market.T,1/secondsPerYear), VOLATILITY, RISK_FREE, true);
            this.options -= qty; this.cash += optPrice;
            this.logMsg(`MOMO SELL ${qty} @ ${optPrice.toFixed(0)}`);
        }
    }
}

/******************************************************************************************
 * 5.  Market & Simulation Loop
 ******************************************************************************************/
const secondsPerYear = SECONDS_PER_DAY*365.25;
const bots = [
    new DeltaHedger("DeltaHedgeBot","#0af"),
    new VolArb("VolArbBot","#f90"),
    new Momentum("MomBot","#3c3")
];
const market = {
    price: INITIAL_PRICE,
    history: [INITIAL_PRICE],
    T: DAYS_TO_EXP / 365.25,
    startTime: Date.now(),
    tick: 0,
    volatility: function() {
        if (this.history.length<20) return VOLATILITY;
        const returns = [];
        for (let i=1;i<20;i++) returns.push(Math.log(this.history[this.history.length-20+i]/this.history[this.history.length-20+i-1]));
        const mean = returns.reduce((a,b)=>a+b)/returns.length;
        const varr = returns.map(r=>Math.pow(r-mean,2)).reduce((a,b)=>a+b)/(returns.length-1);
        return Math.sqrt(varr * SECONDS_PER_DAY*365.25); // annualized
    }
};

function step() {
    // 1. evolve price (geometric brownian)
    const dt = 1 / SECONDS_PER_DAY;
    const drift = RISK_FREE - 0.5*VOLATILITY*VOLATILITY;
    const Z = randn();
    market.price *= Math.exp(drift*dt + VOLATILITY*Math.sqrt(dt)*Z);
    market.history.push(market.price);
    if (market.history.length>200) market.history.shift();

    // 2. time decay
    market.T = Math.max(0, (DAYS_TO_EXP*SECONDS_PER_DAY - (Date.now()-market.startTime)/1000) / secondsPerYear);

    // 3. bots act
    bots.forEach(b => b.think());

    // 4. expiry
    if (market.T<=0) {
        bots.forEach(b => {
            const settle = Math.max(market.price - STRIKE, 0);
            const payoff = b.options * settle;
            b.cash += payoff;
            b.logMsg(`EXPIRY settle ${settle.toFixed(0)} total payoff ${payoff.toFixed(0)}`);
            b.options = 0;
            b.shares = 0;
        });
        market.T = DAYS_TO_EXP / 365.25;
        market.startTime = Date.now();
    }

    // 5. render
    bots.forEach(b => b.render());
}

    market.price=INITIAL_PRICE; market.history=[INITIAL_PRICE]; market.tick=0; market.T=DAYS_TO_EXP/365.25; market.startTime=Date.now();
    bots.forEach(b=>{ b.cash=1e6; b.shares=0; b.options=0; b.premium=0; b.log=[]; b.render(); });
};
